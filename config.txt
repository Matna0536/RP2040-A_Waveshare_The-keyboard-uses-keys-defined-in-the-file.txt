# =====================================================================
# CONFIG.TXT — ไฟล์ตั้งค่าปุ่มกด Macro Keypad
# =====================================================================
#
# รูปแบบ:
#   ปุ่ม=ข้อความหรือลิงก์,โหมด[,CHROME][,RUN][,ENTER]
#
# ปุ่ม = พินเดียว "GP0" หรือพินคู่ (กดพร้อมกัน) "GP0+GP1"
#
# คีย์เวิร์ดโหมด (ใช้ตัวพิมพ์ใหญ่เสมอ):
#   PIN/รหัส         -> พิมพ์ตัวเลขล้วนผ่าน numpad (เสถียรทุกภาษาเครื่อง)
#   TEXT/ข้อความ      -> พิมพ์ข้อความภาษาอังกฤษเต็มรูปแบบ
#   TH หรือ THAI/ภาษาไทย -> พิมพ์ข้อความภาษาไทย (ผสมอังกฤษ/สัญลักษณ์ได้ในสตริงเดียว)
#   MC/มายคราฟ        -> เปิดแชท Minecraft (กด T) แล้วพิมพ์คำสั่งเต็ม
#   VOLUME_UP/+volume/เพิ่มเสียง   -> เพิ่มระดับเสียง (กดค้างได้ ทำซ้ำต่อเนื่อง)
#   VOLUME_DOWN/-volume/ลดเสียง   -> ลดระดับเสียง (กดค้างได้ ทำซ้ำต่อเนื่อง)
#   ARROW_UP/ลูกศรขึ้น            -> ปุ่มลูกศรขึ้น (กดค้างได้)
#   ARROW_DOWN/ลูกศรลง            -> ปุ่มลูกศรลง (กดค้างได้)
#   ARROW_LEFT/ลูกศรซ้าย           -> ปุ่มลูกศรซ้าย (กดค้างได้)
#   ARROW_RIGHT/ลูกศรขวา          -> ปุ่มลูกศรขวา (กดค้างได้)
#   PREV_TRACK/เพลงก่อนหน้า        -> ปุ่มเพลงก่อนหน้า
#   PLAY_PAUSE/เล่นหยุด            -> ปุ่มเล่น/หยุดเพลง
#   NEXT_TRACK/เพลงถัดไป           -> ปุ่มเพลงถัดไป
#
# คีย์เวิร์ดเสริม (ใส่ต่อท้ายโหมดได้):
#   CHROME หรือ GOOGLE (ใช้ร่วมกับโหมด TEXT เท่านั้น)
#       -> แปลงข้อความที่พิมพ์ให้กลายเป็น "chrome.exe <ข้อความ/ลิงก์>"
#          เอาไว้เปิดลิงก์ผ่าน Chrome โดยเฉพาะ
#   RUN (ใช้ร่วมกับ TEXT/PIN/THAI/MC ได้)
#       -> เปิดกล่อง Win+R ก่อนพิมพ์ข้อความ แล้วกด Enter ให้อัตโนมัติเสมอ
#          เหมาะกับสั่งรันคำสั่ง Windows เช่น shutdown, notepad, calc
#   ENTER
#       -> กด Enter ต่อท้ายหลังพิมพ์เสร็จ
#
# =====================================================================
#
# ระบบ LED State (4 สถานะ):
#   State 0 (ทั้ง 3 LED ดับ): ใช้ config แบบปกติ (GP1=...)
#   State 1 (LED1 สว่าง):      ใช้ config GP1_L1=...
#   State 2 (LED2 สว่าง):      ใช้ config GP1_L2=...
#   State 3 (LED3 สว่าง):      ใช้ config GP1_L3=...
#
# ปุ่ม L+ (GP10): State 0→1→2→3→0 (วนรอบ)
# ปุ่ม L- (GP9):  State 0→3→2→1→0 (วนรอบ)
#
# =====================================================================

# ตั้งค่า LED pins (anode ต่อขา GPIO)
LED_PINS=GP13,GP12,GP11

# ตั้งค่าปุ่ม L+/L-
BUTTON_L_PLUS=GP10
BUTTON_L_MINUS=GP9
